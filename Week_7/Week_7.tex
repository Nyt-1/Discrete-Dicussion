\section*{Counting}
Combinations, Permutations, Combinations with Replacement

\begin{enumerate}
  \item[1.] How many distinct ways to form 2 pairs?
  \item[2.] How many ways to rearrange the letters of the word COFFEE?
  \item[3.] Given a collection of 12 students. How many different teams of 3 can we make?
  \item[4.] At a supermarket, Sally wants to buy 12 pieces of fruit. She is able to pick from oranges, pears, and apples. How many different ways is there is buy 12 pieces of fruit?
\end{enumerate}

\section*{Binomial Theorem}
Vandermonde, Pascal's Identity.

\begin{enumerate}
  \item[1.] Calculate using Pascal's Identity:
  \[
  \binom{11}{4}
  \]

  \[
  \binom{12}{5}
  \]

  \[
  \binom{15}{2}
  \]

  \item[2.] Calculate using Vandermonde's Identity
  \[
  \sum_{k=0}^{6}\binom{8}{k}\binom{5}{6-k}
  \]

  \[
  \sum_{k=0}^{4}\binom{9}{k}\binom{7}{4-k}
  \]

  \[
  \sum_{k\in\mathbb{Z}}\binom{12}{k}\binom{9}{10-k}
  \]
\end{enumerate}
